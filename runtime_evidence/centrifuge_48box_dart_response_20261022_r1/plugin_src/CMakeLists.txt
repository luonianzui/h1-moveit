cmake_minimum_required(VERSION 3.16)
project(h12_dart_response_r1)
find_package(ament_cmake REQUIRED)
find_package(gz_dartsim_vendor REQUIRED)
find_package(DART 6.13.2 EXACT REQUIRED CONFIG)
add_library(DartContactResponse SHARED ContactResponse.cc)
target_compile_features(DartContactResponse PRIVATE cxx_std_17)
target_compile_definitions(DartContactResponse PRIVATE NDEBUG)
target_compile_options(DartContactResponse PRIVATE -O3 -fno-fast-math -ffp-contract=off)
target_link_libraries(DartContactResponse PRIVATE dart dl)
add_executable(qualify_response QualifyResponse.cc)
target_compile_features(qualify_response PRIVATE cxx_std_17)
target_compile_definitions(qualify_response PRIVATE NDEBUG)
target_compile_options(qualify_response PRIVATE -O3 -fno-fast-math -ffp-contract=off)
target_link_libraries(qualify_response PRIVATE dart dl)
install(TARGETS DartContactResponse qualify_response LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
ament_package()
