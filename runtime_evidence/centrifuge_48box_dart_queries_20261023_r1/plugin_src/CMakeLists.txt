cmake_minimum_required(VERSION 3.16)
project(h12_dart_queries_r1)
find_package(ament_cmake REQUIRED)
find_package(gz_dartsim_vendor REQUIRED)
find_package(DART 6.13.2 EXACT REQUIRED CONFIG)
add_library(DartConstraintCache SHARED CachedSolver.cc)
target_compile_features(DartConstraintCache PRIVATE cxx_std_17)
target_compile_definitions(DartConstraintCache PRIVATE NDEBUG)
target_compile_options(DartConstraintCache PRIVATE -O3 -fno-fast-math -ffp-contract=off)
target_link_libraries(DartConstraintCache PRIVATE dart dl)
add_executable(qualify_solver QualifySolver.cc)
target_compile_features(qualify_solver PRIVATE cxx_std_17)
target_compile_definitions(qualify_solver PRIVATE NDEBUG)
target_compile_options(qualify_solver PRIVATE -O3 -fno-fast-math -ffp-contract=off)
target_link_libraries(qualify_solver PRIVATE dart dl)
install(TARGETS DartConstraintCache qualify_solver LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
add_executable(qualify_queries QualifyQueries.cc)
target_compile_features(qualify_queries PRIVATE cxx_std_17)
target_compile_definitions(qualify_queries PRIVATE NDEBUG)
target_compile_options(qualify_queries PRIVATE -O3 -fno-fast-math -ffp-contract=off)
target_link_libraries(qualify_queries PRIVATE dart dl)
install(TARGETS qualify_queries RUNTIME DESTINATION bin)
ament_package()
