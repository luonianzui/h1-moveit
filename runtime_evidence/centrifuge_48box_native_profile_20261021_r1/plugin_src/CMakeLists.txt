cmake_minimum_required(VERSION 3.16)
project(h12_task_cpu_profile_r1)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
set(PROFILER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../runtime_dependencies/gperftools_profile_20261015_r1/extracted/usr")
add_library(TaskCpuProfile SHARED TaskCpuProfile.cc)
target_compile_features(TaskCpuProfile PRIVATE cxx_std_17)
target_include_directories(TaskCpuProfile PRIVATE "${PROFILER_ROOT}/include")
target_link_libraries(TaskCpuProfile PRIVATE gz-sim8::gz-sim8 "${PROFILER_ROOT}/lib/x86_64-linux-gnu/libprofiler.so")
set_target_properties(TaskCpuProfile PROPERTIES INSTALL_RPATH "${PROFILER_ROOT}/lib/x86_64-linux-gnu")
install(TARGETS TaskCpuProfile LIBRARY DESTINATION lib)
ament_package()
